CREATE TABLE IF NOT EXISTS exp3_department (
    dept_id INT PRIMARY KEY,
    department_name VARCHAR(50) NOT NULL UNIQUE,
    location VARCHAR(50) NOT NULL
);

CREATE TABLE IF NOT EXISTS exp3_project (
    project_id INT PRIMARY KEY,
    project_name VARCHAR(80) NOT NULL UNIQUE,
    budget DECIMAL(12, 2) NOT NULL
);

CREATE TABLE IF NOT EXISTS exp3_employee (
    emp_id INT PRIMARY KEY,
    full_name VARCHAR(80) NOT NULL,
    dept_id INT NOT NULL,
    designation VARCHAR(50) NOT NULL,
    salary DECIMAL(10, 2) NOT NULL,
    hire_date DATE NOT NULL,
    CONSTRAINT fk_exp3_employee_department
        FOREIGN KEY (dept_id) REFERENCES exp3_department(dept_id)
);

CREATE TABLE IF NOT EXISTS exp3_employee_project (
    emp_id INT NOT NULL,
    project_id INT NOT NULL,
    project_role VARCHAR(60) NOT NULL,
    assigned_on DATE NOT NULL,
    PRIMARY KEY (emp_id, project_id),
    CONSTRAINT fk_exp3_ep_employee
        FOREIGN KEY (emp_id) REFERENCES exp3_employee(emp_id),
    CONSTRAINT fk_exp3_ep_project
        FOREIGN KEY (project_id) REFERENCES exp3_project(project_id)
);

SELECT emp_id, full_name, designation, salary FROM exp3_employee WHERE salary >= 70000 ORDER BY salary DESC;
SELECT full_name, designation FROM exp3_employee ORDER BY full_name;
SELECT d.department_name, COUNT(e.emp_id) AS employee_count, ROUND(AVG(e.salary),2) AS average_salary, MIN(e.salary) AS minimum_salary, MAX(e.salary) AS maximum_salary FROM exp3_department d JOIN exp3_employee e ON e.dept_id=d.dept_id GROUP BY d.dept_id,d.department_name ORDER BY average_salary DESC;
SELECT d.department_name, COUNT(e.emp_id) AS employee_count, ROUND(AVG(e.salary),2) AS average_salary FROM exp3_department d JOIN exp3_employee e ON e.dept_id=d.dept_id GROUP BY d.dept_id,d.department_name HAVING COUNT(e.emp_id)>=6 AND AVG(e.salary)>60000 ORDER BY average_salary DESC;
SELECT full_name,salary,CASE WHEN salary>=90000 THEN 'High' WHEN salary>=65000 THEN 'Medium' ELSE 'Entry' END AS salary_band FROM exp3_employee ORDER BY salary DESC,full_name;
SELECT p.project_name,COUNT(ep.emp_id) AS assigned_employee_count,p.budget FROM exp3_project p LEFT JOIN exp3_employee_project ep ON ep.project_id=p.project_id GROUP BY p.project_id,p.project_name,p.budget ORDER BY assigned_employee_count DESC,p.project_name;
SELECT e.full_name,d.department_name,p.project_name,ep.project_role FROM exp3_employee e JOIN exp3_department d ON d.dept_id=e.dept_id JOIN exp3_employee_project ep ON ep.emp_id=e.emp_id JOIN exp3_project p ON p.project_id=ep.project_id ORDER BY d.department_name,e.full_name,p.project_name;
