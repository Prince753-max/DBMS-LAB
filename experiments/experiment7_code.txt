   Self-contained teaching dataset; uses only exp7_ tables.
   Business rules: one customer per order; one current unit price and supplier
   per product; each product occurs at most once in an order.

   Functional dependencies (FDs):
   order_id -> order_date, customer_id
   customer_id -> customer_name, customer_city
   product_id -> product_name, unit_price, supplier_id
   supplier_id -> supplier_name
   (order_id, product_id) -> quantity
   Candidate key of Sales_1NF: (order_id, product_id).
   These are assumed business rules, supported but not proved by sample rows.
*/

    customer_id INT NOT NULL, customer_name VARCHAR(60) NOT NULL,
    customer_city VARCHAR(40) NOT NULL, items JSON NOT NULL
);

SELECT '1. UNF: repeating product group' AS step;
SELECT order_id, customer_name, JSON_LENGTH(items) AS repeating_items
FROM exp7_sales_unf ORDER BY order_id;

-- 2. 1NF: expand the repeating group into atomic line-item rows.
CREATE TABLE exp7_sales_1nf AS
SELECT u.order_id,u.order_date,u.customer_id,u.customer_name,u.customer_city,
       j.product_id,j.product_name,j.unit_price,j.supplier_id,j.supplier_name,j.quantity
FROM exp7_sales_unf u
JOIN JSON_TABLE(u.items,'$[*]' COLUMNS (
 product_id INT PATH '$.id', product_name VARCHAR(60) PATH '$.name',
 unit_price DECIMAL(10,2) PATH '$.price', supplier_id INT PATH '$.supplier',
 supplier_name VARCHAR(60) PATH '$.supplier_name', quantity INT PATH '$.qty'
)) AS j ON TRUE;
ALTER TABLE exp7_sales_1nf ADD PRIMARY KEY(order_id,product_id);
SELECT '2. 1NF: atomic rows, composite key' AS step;
SELECT * FROM exp7_sales_1nf ORDER BY order_id,product_id;

-- 3. 2NF: remove dependencies on only one part of the composite key.
-- Customer and supplier details are still transitive dependencies here.
CREATE TABLE exp7_orders_2nf AS
SELECT DISTINCT order_id,order_date,customer_id,customer_name,customer_city
FROM exp7_sales_1nf;
ALTER TABLE exp7_orders_2nf ADD PRIMARY KEY(order_id);
CREATE TABLE exp7_products_2nf AS
SELECT DISTINCT product_id,product_name,unit_price,supplier_id,supplier_name
FROM exp7_sales_1nf;
ALTER TABLE exp7_products_2nf ADD PRIMARY KEY(product_id);
CREATE TABLE exp7_lines_2nf AS
SELECT order_id,product_id,quantity FROM exp7_sales_1nf;
ALTER TABLE exp7_lines_2nf ADD PRIMARY KEY(order_id,product_id);
SELECT '3. 2NF: partial dependencies removed' AS step;
SELECT * FROM exp7_orders_2nf ORDER BY order_id;
SELECT * FROM exp7_products_2nf ORDER BY product_id;
SELECT * FROM exp7_lines_2nf ORDER BY order_id,product_id;

-- 4. 3NF: separate customer and supplier transitive dependencies.
CREATE TABLE exp7_customers (
 customer_id INT PRIMARY KEY, customer_name VARCHAR(60) NOT NULL,
 customer_city VARCHAR(40) NOT NULL
);
CREATE TABLE exp7_suppliers (
 supplier_id INT PRIMARY KEY, supplier_name VARCHAR(60) NOT NULL
);
CREATE TABLE exp7_orders (
 order_id INT PRIMARY KEY, order_date DATE NOT NULL, customer_id INT NOT NULL,
 FOREIGN KEY(customer_id) REFERENCES exp7_customers(customer_id)
);
CREATE TABLE exp7_products (
 product_id INT PRIMARY KEY, product_name VARCHAR(60) NOT NULL,
 unit_price DECIMAL(10,2) NOT NULL, supplier_id INT NOT NULL,
 FOREIGN KEY(supplier_id) REFERENCES exp7_suppliers(supplier_id)
);
CREATE TABLE exp7_lines (
 order_id INT NOT NULL, product_id INT NOT NULL, quantity INT NOT NULL,
 PRIMARY KEY(order_id,product_id),
 FOREIGN KEY(order_id) REFERENCES exp7_orders(order_id),
 FOREIGN KEY(product_id) REFERENCES exp7_products(product_id)
);

-- 5. BCNF: no further split is required for this FD set.
-- In Customers, Suppliers, Orders, Products and Lines, every nontrivial
-- determinant is respectively customer_id, supplier_id, order_id, product_id,
-- or (order_id,product_id), each a key of its own relation.
-- Lossless proof: each binary split intersects at a determinant which is a
-- key of the extracted relation: order_id, product_id, customer_id, supplier_id.
-- Dependency preservation: every original FD is enforced within one final
-- relation; checking the FDs requires no join. Keys/FKs enforce structure.

-- 6. Reconstruct all columns and verify BOTH directions, not just row counts.
CREATE VIEW exp7_reconstructed AS
SELECT o.order_id,o.order_date,c.customer_id,c.customer_name,c.customer_city,
 p.product_id,p.product_name,p.unit_price,s.supplier_id,s.supplier_name,l.quantity
FROM exp7_lines l JOIN exp7_orders o ON o.order_id=l.order_id
JOIN exp7_customers c ON c.customer_id=o.customer_id
JOIN exp7_products p ON p.product_id=l.product_id
JOIN exp7_suppliers s ON s.supplier_id=p.supplier_id;
SELECT '5. Lossless join verification: all differences must be zero' AS step;
SELECT (SELECT COUNT(*) FROM exp7_sales_1nf) AS original_rows,
       (SELECT COUNT(*) FROM exp7_reconstructed) AS reconstructed_rows;
SELECT COUNT(*) AS missing_or_changed_rows FROM exp7_sales_1nf a
LEFT JOIN exp7_reconstructed b ON a.order_id=b.order_id AND a.product_id=b.product_id
WHERE b.order_id IS NULL OR NOT (
 a.order_date <=> b.order_date AND a.customer_id <=> b.customer_id AND
 a.customer_name <=> b.customer_name AND a.customer_city <=> b.customer_city AND
 a.product_name <=> b.product_name AND a.unit_price <=> b.unit_price AND
 a.supplier_id <=> b.supplier_id AND a.supplier_name <=> b.supplier_name AND
 a.quantity <=> b.quantity);
SELECT COUNT(*) AS extra_rows FROM exp7_reconstructed b
LEFT JOIN exp7_sales_1nf a ON a.order_id=b.order_id AND a.product_id=b.product_id
WHERE a.order_id IS NULL;

-- 7. SQL checks for each declared FD on the source sample.
SELECT '6. FD verification: every violation count must be zero' AS step;
SELECT 'order_id -> date, customer' AS functional_dependency, COUNT(*) AS violations
FROM (SELECT order_id FROM exp7_sales_1nf GROUP BY order_id
 HAVING COUNT(DISTINCT order_date)>1 OR COUNT(DISTINCT customer_id)>1) v
UNION ALL
SELECT 'customer_id -> name, city',COUNT(*) FROM
 (SELECT customer_id FROM exp7_sales_1nf GROUP BY customer_id
 HAVING COUNT(DISTINCT customer_name)>1 OR COUNT(DISTINCT customer_city)>1) v
UNION ALL
SELECT 'product_id -> name, price, supplier',COUNT(*) FROM
 (SELECT product_id FROM exp7_sales_1nf GROUP BY product_id
 HAVING COUNT(DISTINCT product_name)>1 OR COUNT(DISTINCT unit_price)>1
 OR COUNT(DISTINCT supplier_id)>1) v
UNION ALL
SELECT 'supplier_id -> name',COUNT(*) FROM
 (SELECT supplier_id FROM exp7_sales_1nf GROUP BY supplier_id
 HAVING COUNT(DISTINCT supplier_name)>1) v
UNION ALL
SELECT '(order_id,product_id) -> quantity',COUNT(*) FROM
 (SELECT order_id,product_id FROM exp7_sales_1nf GROUP BY order_id,product_id
 HAVING COUNT(DISTINCT quantity)>1) v;
SELECT order_id,SUM(unit_price*quantity) AS order_total
FROM exp7_reconstructed GROUP BY order_id ORDER BY order_id;
